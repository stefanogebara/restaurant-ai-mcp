
def extract_features(reservation_data):
    """
    Extract 16 features from reservation data for XGBoost model
    
    Features (matching training data):
    1. booking_lead_time_hours - hours between booking and reservation
    2. month_num - month of reservation (1-12)
    3. is_weekend - 1 if Fri/Sat, 0 otherwise
    4. is_prime_time - 1 if 6-9 PM, 0 otherwise
    5. party_size - number of guests
    6. is_tourist - 1 if lead time > 7 days (placeholder)
    7. is_travel_agency - 0 (not available for restaurants)
    8. is_repeat_customer - 0 (requires customer history)
    9. customer_no_show_rate - 0.37 (dataset average)
    10. has_special_requests - 1 if special requests exist
    11. special_request_count - count of special requests
    12. booking_changes_count - 0 (not available)
    13. has_deposit - 0 (not implemented yet)
    14. has_waiting_list - 0 (not implemented yet)
    15. stays_in_weekend_nights - 0 (N/A for restaurants)
    16. stays_in_week_nights - 0 (N/A for restaurants)
    """
    # Parse dates
    reservation_date = datetime.fromisoformat(reservation_data['reservation_date'].replace('Z', '+00:00'))
    booking_date = datetime.fromisoformat(reservation_data.get('booking_date', datetime.now().isoformat()).replace('Z', '+00:00'))
    
    # Calculate time-based features
    lead_time_hours = (reservation_date - booking_date).total_seconds() / 3600
    month_num = reservation_date.month
    day_of_week = reservation_date.weekday()  # 0=Monday, 6=Sunday
    hour = reservation_date.hour
    
    # Calculate derived features
    is_weekend = 1 if day_of_week >= 4 else 0  # Fri-Sun
    is_prime_time = 1 if 18 <= hour <= 21 else 0  # 6-9 PM
    party_size = reservation_data.get('party_size', 2)
    is_tourist = 1 if lead_time_hours > (7 * 24) else 0  # Booked > 7 days ahead
    
    # Special requests features
    special_requests = reservation_data.get('special_requests', '')
    has_special_requests = 1 if special_requests else 0
    special_request_count = len(special_requests.split(',')) if special_requests else 0
    
    # Create feature dictionary with exact feature names
    features = {
        'booking_lead_time_hours': lead_time_hours,
        'month_num': month_num,
        'is_weekend': is_weekend,
        'is_prime_time': is_prime_time,
        'party_size': party_size,
        'is_tourist': is_tourist,
        'is_travel_agency': 0,  # Not available
        'is_repeat_customer': 0,  # Would need customer history
        'customer_no_show_rate': 0.37,  # Dataset average
        'has_special_requests': has_special_requests,
        'special_request_count': special_request_count,
        'booking_changes_count': 0,  # Not tracked yet
        'has_deposit': 0,  # Not implemented
        'has_waiting_list': 0,  # Not implemented
        'stays_in_weekend_nights': 0,  # N/A for restaurants
        'stays_in_week_nights': 0  # N/A for restaurants
    }
    
    # Return as 2D array in correct order
    feature_order = [
        'booking_lead_time_hours', 'month_num', 'is_weekend', 'is_prime_time',
        'party_size', 'is_tourist', 'is_travel_agency', 'is_repeat_customer',
        'customer_no_show_rate', 'has_special_requests', 'special_request_count',
        'booking_changes_count', 'has_deposit', 'has_waiting_list',
        'stays_in_weekend_nights', 'stays_in_week_nights'
    ]
    
    feature_values = [features[f] for f in feature_order]
    return np.array(feature_values, dtype=np.float32).reshape(1, -1), feature_order
